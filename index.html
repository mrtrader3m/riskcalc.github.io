    function calculateRisk() {

      const balance = parseFloat(document.getElementById('accountBalance').value) || 0;

      const riskPct = parseFloat(document.getElementById('riskPercent').value) || 0;

      const riskCashInput = parseFloat(document.getElementById('riskCash').value) || 0;

      const entry = parseFloat(document.getElementById('entryPrice').value) || 0;

      const sl = parseFloat(document.getElementById('stopLossPrice').value) || 0;

      const leverage = parseFloat(document.getElementById('leverageSelect').value) || 200;

      const fxRate = parseFloat(document.getElementById('usdFxRate').value) || 1.0;

      const symbol = currencySymbols[currentCurrency] || '$';

 

      // Determine Risk Cash in local base currency

      let riskCashLocal = 0;

      if (riskMode === 'percent') {

        riskCashLocal = balance * (riskPct / 100);

        document.getElementById('riskCash').value = riskCashLocal.toFixed(2);

      } else {

        riskCashLocal = riskCashInput;

        const computedPct = balance > 0 ? (riskCashLocal / balance) * 100 : 0;

        document.getElementById('riskPercent').value = computedPct.toFixed(2);

      }

 

      // Convert local risk cash to USD

      const riskCashUSD = riskCashLocal / fxRate;

 

      // Stop Loss distance in USD and pips (1 pip = $0.10 in gold)

      const slDistUSD = Math.abs(entry - sl);

      const slPips = slDistUSD * 10; // 1.00 move = 10 pips

 

      document.getElementById('slPipsDisplay').textContent = `${slPips.toFixed(0)} pips ($${slDistUSD.toFixed(2)})`;

 

      if (slDistUSD <= 0 || balance <= 0) {

        document.getElementById('resLots').textContent = '0.00';

        return;

      }

 

      // Gold Standard Lot = 100 oz

      // Loss per standard lot = slDistUSD * 100

      const lossPerLotUSD = slDistUSD * 100;

      const lots = riskCashUSD / lossPerLotUSD;

 

      const miniLots = lots * 10;

      const microLots = lots * 100;

 

      // Render Lot Output

      document.getElementById('resLots').textContent = lots.toFixed(2);

      document.getElementById('resMiniLots').textContent = miniLots.toFixed(2);

      document.getElementById('resMicroLots').textContent = microLots.toFixed(2);

 

      // Render Max Risk Output

      document.getElementById('resMaxRiskCash').textContent = `${symbol} ${riskCashLocal.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2})}`;

      document.getElementById('resMaxRiskPct').textContent = `${((riskCashLocal/balance)*100).toFixed(2)}% Account`;

 

      // Pip value per standard lot in Local Base Currency

      // 1 pip ($0.10 move) on 1 standard lot (100 oz) = $10 USD

      const pipValueStandardUSD = 10;

      const pipValueTotalLocal = (lots * pipValueStandardUSD) * fxRate;

      document.getElementById('resPipValue').textContent = `${symbol} ${pipValueTotalLocal.toFixed(2)}`;

 

      // Required Margin Calculation

      // Notional USD = Lots * 100 oz * Entry Price

      const totalOz = lots * 100;

      const notionalUSD = totalOz * entry;

      const notionalLocal = notionalUSD * fxRate;

      const reqMarginUSD = notionalUSD / leverage;

      document.getElementById('resNotionalVal').textContent = `$${notionalUSD.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2})}`;

      document.getElementById('resReqMargin').textContent = `${symbol} ${reqMarginLocal.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2})}`;

      document.getElementById('resFreeMarginPct').textContent = `${((reqMarginLocal/balance)*100).toFixed(2)}% Used`;

      document.getElementById('resEffectiveLeverage').textContent = `1:${(notionalLocal / balance).toFixed(2)}`;

      document.getElementById('resPipsDistance').textContent = `${slPips.toFixed(0)} pips ($${slDistUSD.toFixed(2)})`;

 

      // Take Profit Partial Exits Calculation

      const tp1P = parseFloat(document.getElementById('tp1Price').value) || 0;

      const tp1Pct = parseFloat(document.getElementById('tp1Pct').value) || 0;

        const targetPips = distUSD * 10;

        const targetLots = lots * (t.pct / 100);

        const profitUSD = (distUSD * 100) * targetLots;

        const profitLocal = profitUSD * fxRate;

        const rr = slDistUSD > 0 ? (distUSD / slDistUSD) : 0;

    function calculateCompounding() {

      const startBal = parseFloat(document.getElementById('cmpStartBalance').value) || 0;

      const ratePct = parseFloat(document.getElementById('cmpReturnPct').value) || 0;

      const periods = parseInt(document.getElementById('cmpPeriods').value) || 1;

      const reinvestPct = parseFloat(document.getElementById('cmpReinvestPct').value) || 100;

      const monthlyAdd = parseFloat(document.getElementById('cmpMonthlyAdd').value) || 0;

      const symbol = currencySymbols[currentCurrency] || '$';

 

      let currentBal = startBal;

      let totalProfitAccum = 0;

      let totalPayoutsAccum = 0;

 

      const labels = ['Start'];

      const dataPoints = [startBal];

      let tableRowsHTML = '';

 

      for (let p = 1; p <= periods; p++) {

        const periodStart = currentBal;

        const grossProfit = periodStart * (ratePct / 100);

        const reinvestedProfit = grossProfit * (reinvestPct / 100);

        const payoutProfit = grossProfit - reinvestedProfit;

 

        currentBal = periodStart + reinvestedProfit + monthlyAdd;

        totalProfitAccum += grossProfit;

        totalPayoutsAccum += payoutProfit;

 

        labels.push(`P${p}`);

        dataPoints.push(currentBal);

 

        tableRowsHTML += `

          <tr class="hover:bg-gray-800/40">

            <td class="py-1.5 px-2 font-bold text-amber-400">${p}</td>

            <td class="py-1.5 px-2">${symbol} ${periodStart.toFixed(0)}</td>

            <td class="py-1.5 px-2 text-emerald-400 font-semibold">+${symbol} ${grossProfit.toFixed(0)}</td>

            <td class="py-1.5 px-2 font-bold text-white">${symbol} ${currentBal.toFixed(0)}</td>

          </tr>

        `;

      }

 

      document.getElementById('cmpTableBody').innerHTML = tableRowsHTML;

      document.getElementById('cmpResEndBalance').textContent = `${symbol} ${currentBal.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2})}`;

      document.getElementById('cmpResTotalProfit').textContent = `${symbol} ${totalProfitAccum.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2})}`;

      document.getElementById('cmpResTotalPayouts').textContent = `${symbol} ${totalPayoutsAccum.toLocaleString(undefined, {minimumFractionDigits: 2, maximumFractionDigits: 2})}`;

      renderCompoundingChart(labels, dataPoints);

    }

    function calcCustomDrawdown() {

      const peak = parseFloat(document.getElementById('ddPeakEquity').value) || 0;

      const current = parseFloat(document.getElementById('ddCurrentEquity').value) || 0;

      const symbol = currencySymbols[currentCurrency] || '$';

      const lossPct = peak > 0 ? (loss / peak) * 100 : 0;

      const requiredPct = current > 0 ? (loss / current) * 100 : 0;

 

      document.getElementById('ddResultCash').textContent = `${symbol} ${loss.toLocaleString(undefined, {minimumFractionDigits: 2})} (-${lossPct.toFixed(2)}%)`;

      document.getElementById('ddResultRequiredPct').textContent = `+${requiredPct.toFixed(2)}%`;

    }

    function renderJournalTable() {

      const journal = getJournal();

      const body = document.getElementById('journalTableBody');

 

      if (journal.length === 0) {

        body.innerHTML = `<tr><td colspan="9" class="text-center py-6 text-gray-500">No saved trade setups in journal.</td></tr>`;

        return;

      }

 

      let html = '';

      journal.forEach(t => {

        html += `

          <tr class="hover:bg-gray-800/50">

            <td class="py-2.5 px-3 text-gray-400 text-[11px]">${t.date}</td>

            <td class="py-2.5 px-3"><span class="px-2 py-0.5 rounded font-bold text-[10px] ${t.type === 'BUY' ? 'bg-emerald-500/20 text-emerald-400' : 'bg-rose-500/20 text-rose-400'}">${t.type}</span></td>

            <td class="py-2.5 px-3 font-bold text-white">$${t.entry.toFixed(2)}</td>

            <td class="py-2.5 px-3 text-rose-400">$${t.sl.toFixed(2)}</td>

            <td class="py-2.5 px-3 text-emerald-400">$${t.tp1.toFixed(2)}</td>

            <td class="py-2.5 px-3 font-bold text-amber-400">${t.lots}</td>

            <td class="py-2.5 px-3 text-rose-400">${t.risk}</td>

            <td class="py-2.5 px-3 font-bold text-emerald-400">${t.profit}</td>

            <td class="py-2.5 px-3 text-right">

              <button onclick="deleteJournalItem(${t.id})" class="text-gray-500 hover:text-rose-400"><i data-lucide="trash" class="w-3.5 h-3.5"></i></button>

            </td>

          </tr>

        `;

      });

 

      body.innerHTML = html;

      lucide.createIcons();

    }

 

    function deleteJournalItem(id) {

      let journal = getJournal();

      journal = journal.filter(i => i.id !== id);

      saveJournal(journal);

      renderJournalTable();

    }

 

    function clearJournal() {

      if (confirm('Clear all saved journal entries?')) {

        renderJournalTable();

      }

    }

      const blob = new Blob([csv], { type: 'text/csv' });

      const a = document.createElement('a');

      a.href = url;

      a.download = `MAR_XAUUSD_Journal_${Date.now()}.csv`;

      a.click();
